**Goal:** Add a lightweight tagging system with first-class tag entities (name + description), a join table for many-to-many task-tag relationships, and full MCP + CLI support.

**Tech Stack:** rust-query 0.8 (schema migration, Select derive, delete via downgrade), rmcp (MCP tools), clap (CLI)

```rust
use anyhow::Result;
use rust_query::{optional, Select};

use crate::schema;
use crate::types::{TagData, TagInfo, TaskId};
use crate::Db;

#[derive(Select)]
struct TagSelect {
    name: String,
    description: String,
}

impl Db {
    /// Add a tag to a task. Auto-creates the tag if it doesn't exist. Idempotent.
    pub fn tag_task(&self, task_id: TaskId, tag_name: &str) -> Result<()> {
        // Check if already tagged (via join query)
        let already_tagged = !self.database.transaction(|txn| {
            txn.query(|q| {
                let task = q.join(schema::Task);
                let tt = q.join(schema::TaskTag);
                let tag = q.join(schema::Tag);
                q.filter(task.external_id.eq(task_id));
                q.filter(tt.task.eq(&task));
                q.filter(tt.tag.eq(&tag));
                q.filter(tag.name.eq(tag_name));
                q.into_vec(&task.external_id)
            })
        }).is_empty();

        if already_tagged {
            return Ok(());
        }

        self.database.transaction_mut(|txn| {
            // Verify task exists
            let task_row = txn
                .query_one(optional(|row| {
                    let task = row.and(schema::Task.external_id(task_id));
                    row.then(task)
                }))
                .ok_or_else(|| anyhow::anyhow!("task not found: {task_id}"))?;

            // Find or create tag
            let tag_row = match txn.query_one(optional(|row| {
                let t = row.and(schema::Tag.name(tag_name));
                row.then(t)
            })) {
                Some(row) => row,
                None => txn
                    .insert(schema::Tag {
                        name: tag_name.to_owned(),
                        description: String::new(),
                    })
                    .map_err(|_| anyhow::anyhow!("failed to create tag"))?,
            };

            // Insert task-tag association
            txn.insert_ok(schema::TaskTag {
                task: task_row,
                tag: tag_row,
            });

            Ok(())
        })
    }

    /// Remove a tag from a task.
    pub fn untag_task(&self, task_id: TaskId, tag_name: &str) -> Result<()> {
        self.database.transaction_mut(|txn| {
            // Find the task-tag row via join
            let task_tag_rows: Vec<_> = txn.query(|q| {
                let task = q.join(schema::Task);
                let tt = q.join(schema::TaskTag);
                let tag = q.join(schema::Tag);
                q.filter(task.external_id.eq(task_id));
                q.filter(tt.task.eq(&task));
                q.filter(tt.tag.eq(&tag));
                q.filter(tag.name.eq(tag_name));
                q.into_vec(tt)
            });

            if let Some(row) = task_tag_rows.into_iter().next() {
                let txn = txn.downgrade();
                let _ = txn.delete(row);
            }

            Ok(())
        })
    }

    /// Load tags for a given task.
    pub fn load_tags_for_task(&self, task_id: TaskId) -> Result<Vec<TagData>> {
        let tags = self.database.transaction(|txn| {
            txn.query(|q| {
                let task = q.join(schema::Task);
                let tt = q.join(schema::TaskTag);
                let tag = q.join(schema::Tag);
                q.filter(task.external_id.eq(task_id));
                q.filter(tt.task.eq(&task));
                q.filter(tt.tag.eq(&tag));
                q.into_vec(TagSelectSelect {
                    name: &tag.name,
                    description: &tag.description,
                })
            })
        });

        Ok(tags
            .into_iter()
            .map(|t| TagData {
                name: t.name,
                description: t.description,
            })
            .collect())
    }

    /// List all tags with task counts. If workspace is given, only include tags
    /// that have at least one non-deleted task in that workspace.
    pub fn list_tags(&self, workspace: Option<&str>) -> Result<Vec<TagInfo>> {
        // Get all tags
        let all_tags: Vec<TagSelect> = self.database.transaction(|txn| {
            txn.query(|q| {
                let tag = q.join(schema::Tag);
                q.into_vec(TagSelectSelect {
                    name: &tag.name,
                    description: &tag.description,
                })
            })
        });

        // Get all task-tag pairs with workspace + deleted info
        #[derive(Select)]
        struct TaskTagInfo {
            tag_name: String,
            task_workspace: String,
            task_deleted_at: Option<String>,
        }

        let pairs: Vec<TaskTagInfo> = self.database.transaction(|txn| {
            txn.query(|q| {
                let task = q.join(schema::Task);
                let tt = q.join(schema::TaskTag);
                let tag = q.join(schema::Tag);
                q.filter(tt.task.eq(&task));
                q.filter(tt.tag.eq(&tag));
                q.into_vec(TaskTagInfoSelect {
                    tag_name: &tag.name,
                    task_workspace: &task.workspace,
                    task_deleted_at: &task.deleted_at,
                })
            })
        });

        // Count tasks per tag
        let mut counts: std::collections::HashMap<String, usize> = std::collections::HashMap::new();
        for pair in &pairs {
            if pair.task_deleted_at.is_some() {
                continue;
            }
            if let Some(ws) = workspace {
                if pair.task_workspace != ws {
                    continue;
                }
            }
            *counts.entry(pair.tag_name.clone()).or_default() += 1;
        }

        let mut results: Vec<TagInfo> = all_tags
            .into_iter()
            .filter_map(|tag| {
                let count = counts.get(&tag.name).copied().unwrap_or(0);
                // If workspace filter is active, skip tags with 0 tasks in that workspace
                if workspace.is_some() && count == 0 {
                    return None;
                }
                Some(TagInfo {
                    name: tag.name,
                    description: tag.description,
                    task_count: count,
                })
            })
            .collect();

        results.sort_by(|a, b| a.name.cmp(&b.name));
        Ok(results)
    }

    /// Rename a tag and/or update its description.
    pub fn edit_tag(
        &self,
        current_name: &str,
        new_name: Option<&str>,
        description: Option<&str>,
    ) -> Result<()> {
        self.database.transaction_mut(|txn| {
            let tag_row = txn
                .query_one(optional(|row| {
                    let t = row.and(schema::Tag.name(current_name));
                    row.then(t)
                }))
                .ok_or_else(|| anyhow::anyhow!("tag not found: {current_name}"))?;

            let mut tag = txn.mutable(&tag_row);
            if let Some(name) = new_name {
                tag.name = name.to_owned();
            }
            if let Some(desc) = description {
                tag.description = desc.to_owned();
            }

            Ok(())
        })
    }

    /// Delete a tag and all its task associations.
    pub fn delete_tag(&self, tag_name: &str) -> Result<()> {
        self.database.transaction_mut(|txn| {
            // Verify tag exists
            let _tag_exists = txn
                .query_one(optional(|row| {
                    let t = row.and(schema::Tag.name(tag_name));
                    row.then_select(true)
                }))
                .ok_or_else(|| anyhow::anyhow!("tag not found: {tag_name}"))?;

            // Collect all TaskTag rows for this tag
            let task_tag_rows: Vec<_> = txn.query(|q| {
                let tt = q.join(schema::TaskTag);
                let tag = q.join(schema::Tag);
                q.filter(tt.tag.eq(&tag));
                q.filter(tag.name.eq(tag_name));
                q.into_vec(tt)
            });

            // Collect the tag row itself
            let tag_row = txn
                .query_one(optional(|row| {
                    let t = row.and(schema::Tag.name(tag_name));
                    row.then(t)
                }))
                .unwrap(); // We already verified it exists

            // Downgrade transaction to allow deletions
            let txn = txn.downgrade();

            // Delete all task-tag associations first (they reference the tag)
            for row in task_tag_rows {
                let _ = txn.delete(row);
            }

            // Delete the tag itself
            let _ = txn.delete(tag_row);

            Ok(())
        })
    }
}
```

```rust
pub fn list_tasks(&self, filter: TaskFilter) -> Result<Vec<Task>> {
    let rows = TaskSelect::query_all(self);
    let today = Utc::now().date_naive();

    let mut tasks: Vec<Task> = rows
        .into_iter()
        .map(|ts| ts.into_task())
        .collect::<Result<Vec<_>>>()?
        .into_iter()
        .filter(|t| t.deleted_at.is_none())
        .filter(|t| match &filter.status {
            Some(s) => t.status == *s,
            None => true,
        })
        .filter(|t| match &filter.workspace {
            Some(w) => t.workspace == *w,
            None => true,
        })
        .filter(|t| match filter.has_planned_date {
            Some(true) => t.planned_date.is_some(),
            Some(false) => t.planned_date.is_none(),
            None => true,
        })
        .filter(|t| match filter.priority_max {
            Some(max) => t.priority.map_or(false, |p| p <= max),
            None => true,
        })
        .filter(|t| {
            if filter.include_snoozed {
                return true;
            }
            match &t.snooze_until {
                Some(s) => match s.parse::<chrono::NaiveDate>() {
                    Ok(d) => d <= today,
                    Err(_) => true,
                },
                None => true,
            }
        })
        .collect();

    // Sort (existing code unchanged)
    tasks.sort_by(|a, b| { /* ... existing sort ... */ });

    // Populate external contexts
    for task in &mut tasks {
        task.github_pr_contexts = self.load_github_contexts(task.id)?;
        task.linear_contexts = self.load_linear_contexts(task.id)?;
        task.tags = self.load_tags_for_task(task.id)?;
    }

    // Refresh stale contexts
    self.refresh_github_contexts(&mut tasks)?;
    self.refresh_linear_contexts(&mut tasks)?;

    // Apply tag filter (after tags are populated)
    if let Some(ref tag_name) = filter.tag {
        tasks.retain(|t| t.tags.iter().any(|td| td.name == *tag_name));
    }

    Ok(tasks)
}
```

```rust
#[tool(description = "List all tags with task counts. If workspace is given, only shows tags with tasks in that workspace.")]
fn list_tags(
    &self,
    Parameters(params): Parameters<ListTagsParams>,
) -> Result<CallToolResult, McpError> {
    let workspace = params.workspace;

    let result = self.db.run(move |db| {
        let tags = db.list_tags(workspace.as_deref())?;
        if tags.is_empty() {
            return Ok("No tags found.".to_string());
        }
        let lines: Vec<String> = tags
            .iter()
            .map(|t| {
                let desc = if t.description.is_empty() {
                    String::new()
                } else {
                    format!(" — {}", t.description)
                };
                format!("{} ({} tasks){desc}", t.name, t.task_count)
            })
            .collect();
        Ok(lines.join("\n"))
    }).map_err(to_mcp_error)?;

    Ok(CallToolResult::success(vec![Content::text(result)]))
}
```
